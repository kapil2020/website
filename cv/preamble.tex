% --- The macros the whole CV is written in -----------------------------------

% A labelled row: date, identifier or category on the left; content on the right.
\newenvironment{cvrows}[1][5pt]
  {\begin{list}{}{%
      \setlength{\leftmargin}{\cvgap}
      \setlength{\labelwidth}{\cvlab}
      \setlength{\labelsep}{\dimexpr\cvgap-\cvlab\relax}
      \setlength{\itemsep}{#1}
      \setlength{\parsep}{1pt}
      \setlength{\topsep}{3pt}
      \setlength{\partopsep}{0pt}
      \renewcommand{\makelabel}[1]{\hfill\lab ##1}}}
  {\end{list}}

\newcommand{\row}[2]{\item[#1] #2}

% An appointment or degree: role in ink, place beneath, location on the right.
% \post{dates}{role}{institution and detail}{location}
\newcommand{\post}[4]{%
  \item[#1]{\bfseries\color{ink}#2}%
  \ifblank{#4}{}{\hfill\note{#4}}\\[1.5pt]#3}

% One publication.
% \pubentry{id}{authors}{title}{venue}{year}{status}{url}{link text}
\newcommand{\pubentry}[8]{%
  \item[{[#1]}]#2\ \emph{#3}\ {\bfseries\color{venue}#4}%
  \ifblank{#5}{}{{\bfseries\color{venue}, #5}}.%
  \ifblank{#6}{}{\ {\fontsize{7.3}{9}\selectfont\ttfamily\color{accent}[\MakeUppercase{#6}]}}%
  \ifblank{#7}{}{\ {\fontsize{7.5}{9}\selectfont\quietlink{#7}{#8}}}%
}

% Links that should not shout. A DOI is metadata, and a software title is a
% heading that happens to be clickable — neither wants the accent colour that
% hyperref gives every link by default.
\newcommand{\quietlink}[2]{\href{#1}{\textcolor{faint}{#2}}}
\newcommand{\inklink}[2]{\href{#1}{\textcolor{ink}{\bfseries #2}}}

% My own name, bold, so it is findable at a glance in a long author list.
\newcommand{\me}{\textbf{\color{ink}Meena, K.\,K.}}
