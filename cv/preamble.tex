\newcommand{\detail}[1]{{\small\color{muted}#1}}

% Width of the left-hand date/label column. The page hangs off this measure.
\newlength{\cvlab}\setlength{\cvlab}{0.95in}

% --- Section heads ---------------------------------------------------------------
\titleformat{\section}
  {\normalfont\large\scshape\color{ink}}
  {}{0pt}{}
  [\vspace{-4pt}{\color{rulegray}\titlerule[0.4pt]}]
\titlespacing*{\section}{0pt}{14pt plus 3pt minus 2pt}{7pt}

\titleformat{\subsection}
  {\normalfont\normalsize\itshape\color{ink}}
  {}{0pt}{}
\titlespacing*{\subsection}{0pt}{9pt plus 2pt minus 1pt}{3pt}

% --- Macros ---------------------------------------------------------------------

% A list with a label column: dates, or a publication number.
\newenvironment{cvlist}[1][4pt]
  {\begin{list}{}{%
      \setlength{\leftmargin}{\cvlab}
      \setlength{\labelwidth}{\dimexpr\cvlab-10pt\relax}
      \setlength{\labelsep}{10pt}
      \setlength{\itemsep}{#1}
      \setlength{\parsep}{0pt}
      \setlength{\topsep}{2pt}
      \setlength{\partopsep}{0pt}
      \renewcommand{\makelabel}[1]{##1\hfill}}}
  {\end{list}}

\newcommand{\entry}[2]{\item[#1] #2}

% A position or a degree.
% \post{dates}{title}{institution}{place}{detail}
\newcommand{\post}[5]{%
  \item[#1]\textbf{#2}, #3\ifblank{#4}{}{\hfill #4}%
  \ifblank{#5}{}{\\[1pt]\detail{#5}}}

% One publication.
% \pub{id}{authors}{year}{title}{venue}{trailing: status, doi}
\newcommand{\pub}[6]{%
  \item[#1]#2 (#3). #4. \emph{#5}.\ifblank{#6}{}{ #6}}

\newcommand{\doi}[1]{\href{https://doi.org/#1}{\nolinkurl{doi:#1}}}

% My own name, bold, so it can be found at a glance in an author list.
\newcommand{\me}{\textbf{Meena, K.\,K.}}
